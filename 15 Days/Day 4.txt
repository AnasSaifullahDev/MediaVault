      \<p>{type} • {status}\</p>
        {currentProgress !== null && (
          \<p>

            {totalProgress !== null && \` / ${totalProgress}\`}
          \</p>
        )}
      {rating !== null && \<p>Rating: {rating}\</p>}

  )
}
